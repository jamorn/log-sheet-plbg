CREATE TABLE UnitPLBG (
    UnitId INT PRIMARY KEY,

    BaggingCode VARCHAR(10) NOT NULL,  -- เพิ่มตามที่คุยกัน: PL หรือ SA

-- 4. Machines (ปรับตามข้อมูลจริง + Computed Columns)
CREATE TABLE Machines (
    MachineId INT PRIMARY KEY,
    MachineName VARCHAR(10) NOT NULL,
    MachineClass VARCHAR(5) NULL DEFAULT 'g1',
    MachineActive BIT NOT NULL DEFAULT 1,
    MachineLine VARCHAR(1) NULL,
    
    -- Link to Unit (เปลี่ยนจาก CostCenterCode เป็น UnitId โดยตรง)
    UnitId INT NOT NULL,
    
    -- OEE Configuration (ตามข้อมูลจริงที่คุยกัน)
    StandardRPM INT NOT NULL,
    BagWeightKG DECIMAL(5, 2) NOT NULL DEFAULT 25.00,
    
    -- Computed Columns for Performance (เพิ่มประสิทธิภาพ)
    IdealCycleTimeSec AS (60.0 / NULLIF(StandardRPM, 0)) PERSISTED,
    IdealOutputPerHour AS (StandardRPM * 60 * BagWeightKG / 1000.0) PERSISTED, -- KG per hour
    
    -- Maintenance Info
    InstallationDate DATE NULL,
    LastMaintenanceDate DATE NULL,
    NextMaintenanceDate DATE NULL,
    
    -- Audit
    CreatedAt DATETIME DEFAULT GETDATE(),
    UpdatedAt DATETIME DEFAULT GETDATE(),
    
    -- Constraints
    CONSTRAINT FK_Machines_UnitPLBG FOREIGN KEY (UnitId)
        REFERENCES UnitPLBG(UnitId),
    CONSTRAINT CHK_StandardRPM CHECK (StandardRPM > 0),
    CONSTRAINT CHK_BagWeight CHECK (BagWeightKG > 0 AND BagWeightKG <= 50),
    CONSTRAINT UQ_MachineName UNIQUE (MachineName)
);

-- 5. Shifts (เพิ่ม IsCrossMidnight สำหรับกะดึก)
CREATE TABLE Shifts (
    ShiftID INT PRIMARY KEY,
    ShiftName VARCHAR(20) NOT NULL,
    StartTime TIME NOT NULL,
    EndTime TIME NOT NULL,
    IsCrossMidnight BIT NOT NULL DEFAULT 0,  -- 1=ข้ามเที่ยงคืน (กะดึก)
    ShiftHours DECIMAL(4,2) DEFAULT 8.00,
    
    CONSTRAINT CHK_ShiftTime CHECK (EndTime > StartTime OR IsCrossMidnight = 1)
);

-- 11. ProductionRecords (เพิ่ม Function สำหรับกะข้ามเที่ยงคืน)
-- สร้าง Function ก่อน
GO
CREATE FUNCTION dbo.GetShiftDate
(
    @DateTime DATETIME,
    @ShiftID INT
)
RETURNS DATE
AS
BEGIN
    DECLARE @ShiftDate DATE;
    DECLARE @IsCrossMidnight BIT;
    DECLARE @StartTime TIME;
    DECLARE @EndTime TIME;
    
    SELECT 
        @IsCrossMidnight = IsCrossMidnight,
        @StartTime = StartTime,
        @EndTime = EndTime
    FROM Shifts 
    WHERE ShiftID = @ShiftID;
    
    IF @IsCrossMidnight IS NULL
        RETURN CAST(@DateTime AS DATE);
    
    IF @IsCrossMidnight = 0
        SET @ShiftDate = CAST(@DateTime AS DATE);
    ELSE
    BEGIN
        DECLARE @TimeOnly TIME = CAST(@DateTime AS TIME);
        IF @TimeOnly >= '00:00' AND @TimeOnly < @EndTime
            SET @ShiftDate = DATEADD(DAY, -1, CAST(@DateTime AS DATE));
        ELSE
            SET @ShiftDate = CAST(@DateTime AS DATE);
    END
    
    RETURN @ShiftDate;
END
GO

-- ตาราง ProductionRecords
CREATE TABLE ProductionRecords (
    ProductionRecordID BIGINT IDENTITY(1,1) PRIMARY KEY,
    
    -- Basic Information
    MachineId INT NOT NULL,
    ShiftID INT NOT NULL,
    ShiftDate AS dbo.GetShiftDate(SegmentStartDateTime, ShiftID) PERSISTED,
    
    -- Time Information
    SegmentStartDateTime DATETIME NOT NULL,
    SegmentEndDateTime DATETIME NOT NULL,
    SegmentDurationMin AS DATEDIFF(MINUTE, SegmentStartDateTime, SegmentEndDateTime) PERSISTED,
    
    -- Production Data
    BagOutActualMT DECIMAL(10, 3) NOT NULL,
    EstimatedBags AS (BagOutActualMT * 1000 / 
        (SELECT BagWeightKG FROM Machines WHERE MachineId = ProductionRecords.MachineId)) PERSISTED,
    
    -- Status
    RecordStatus VARCHAR(20) NOT NULL DEFAULT 'OPEN'
        CHECK (RecordStatus IN ('OPEN', 'IN_PROGRESS', 'CLOSED', 'CANCELLED')),
    ProductionStatus VARCHAR(20) NULL
        CHECK (ProductionStatus IN ('RUNNING', 'IDLE', 'STOPPED', 'MAINTENANCE')),
    
    -- Audit
    EnteredBy NVARCHAR(128) NOT NULL,
    EnteredAt DATETIME DEFAULT GETDATE(),
    VerifiedBy NVARCHAR(128) NULL,
    VerifiedAt DATETIME NULL,
    LastUpdated DATETIME DEFAULT GETDATE(),
    
    -- Foreign Keys
    CONSTRAINT FK_PR_Machines FOREIGN KEY (MachineId)
        REFERENCES Machines(MachineId),
    CONSTRAINT FK_PR_Shifts FOREIGN KEY (ShiftID)
        REFERENCES Shifts(ShiftID),
    CONSTRAINT FK_PR_EnteredBy FOREIGN KEY (EnteredBy) 
        REFERENCES EmployeeDetails(WindowsLogin),
    CONSTRAINT FK_PR_VerifiedBy FOREIGN KEY (VerifiedBy) 
        REFERENCES EmployeeDetails(WindowsLogin),
    
    -- Check Constraints
    CONSTRAINT CHK_SegmentTime CHECK (SegmentEndDateTime > SegmentStartDateTime),
    CONSTRAINT CHK_BagOutWeight CHECK (BagOutActualMT >= 0),
    
    -- Unique Constraint
    CONSTRAINT UQ_Production_Time_Machine UNIQUE (MachineId, SegmentStartDateTime, SegmentEndDateTime)
);

-- 12. ProblemRecords
CREATE TABLE ProblemRecords (
    ProblemRecordID BIGINT IDENTITY(1,1) PRIMARY KEY,

    -- Foreign Keys
    ProductionRecordID BIGINT NOT NULL,
    ProblemTypeID INT NULL,

    -- Downtime window (clear naming)
    DowntimeStart DATETIME NOT NULL,    -- เวลาเริ่มเหตุการณ์ (เครื่องหยุด)
    DowntimeEnd   DATETIME NULL,        -- เวลาแก้ไข/คืนสภาพ (NULL = ยังไม่เสร็จ)

    -- Computed column (will be NULL until DowntimeEnd is set)
    DowntimeDurationMinutes AS DATEDIFF(MINUTE, DowntimeStart, DowntimeEnd) PERSISTED,

    -- Constraints for validity and max duration
    CONSTRAINT CHK_DowntimeTime CHECK (DowntimeEnd IS NULL OR DowntimeEnd > DowntimeStart),
    CONSTRAINT CHK_MaxDowntime CHECK (DowntimeEnd IS NULL OR DATEDIFF(MINUTE, DowntimeStart, DowntimeEnd) <= 1440),

    -- Problem Details
    IsMachineDowntime BIT NOT NULL DEFAULT 1,
    ProblemDescription NVARCHAR(500) NULL,
    RootCause NVARCHAR(500) NULL,
    ActionTaken NVARCHAR(500) NULL,

    -- Personnel
    OperatorLogin NVARCHAR(128) NULL,
    MaintenanceLogin NVARCHAR(128) NULL,
    AcknowledgedBy NVARCHAR(128) NULL,
    ResolvedBy NVARCHAR(128) NULL,

    -- Status
    ProblemStatus VARCHAR(20) DEFAULT 'OPEN'
        CHECK (ProblemStatus IN ('OPEN', 'IN_PROGRESS', 'RESOLVED', 'ESCALATED')),
    ResolutionTime DATETIME NULL,

    -- Audit / Entry
    EnteredBy NVARCHAR(128) NOT NULL,
    EnteredAt DATETIME NOT NULL DEFAULT GETDATE(),
    CreatedAt DATETIME DEFAULT GETDATE(),
    UpdatedAt DATETIME DEFAULT GETDATE(),

    -- Foreign Keys
    CONSTRAINT FK_PRec_PR FOREIGN KEY (ProductionRecordID)
        REFERENCES ProductionRecords(ProductionRecordID),
    CONSTRAINT FK_PRec_OeePType FOREIGN KEY (ProblemTypeID) 
        REFERENCES OeeLossTypes(ProblemTypeID),
    CONSTRAINT FK_PRec_Operator FOREIGN KEY (OperatorLogin) 
        REFERENCES EmployeeDetails(WindowsLogin),
    CONSTRAINT FK_PRec_Maintenance FOREIGN KEY (MaintenanceLogin) 
        REFERENCES EmployeeDetails(WindowsLogin),
    CONSTRAINT FK_PRec_ResolvedBy FOREIGN KEY (ResolvedBy) 
        REFERENCES EmployeeDetails(WindowsLogin)
);

-- 13. LotCompletionDetails
CREATE TABLE LotCompletionDetails (
    LotDetailID BIGINT IDENTITY(1,1) PRIMARY KEY,
    ProductionRecordID BIGINT NOT NULL,
    
    -- Quality Data
    TotalBags INT NOT NULL,
    GoodBags INT NOT NULL,
    WasteRejectBags INT NOT NULL DEFAULT 0,
    WasteRejectKG DECIMAL(10, 3) NULL,
    
    -- Quality Metrics (Computed Columns)
    FirstPassYield DECIMAL(5,2) AS 
        CASE WHEN TotalBags > 0 THEN (GoodBags * 100.0 / TotalBags) ELSE 0 END PERSISTED,
    RejectRate DECIMAL(5,2) AS 
        CASE WHEN TotalBags > 0 THEN (WasteRejectBags * 100.0 / TotalBags) ELSE 0 END PERSISTED,
    
    -- Lot Information
    LotNumber VARCHAR(50) NULL,
    BaggingCode VARCHAR(10) NULL,
    InspectionBy NVARCHAR(128) NULL,
    InspectionTime DATETIME NULL,
    
    -- Comments
    QualityRemarks NVARCHAR(500) NULL,
    
    -- Foreign Keys
    CONSTRAINT FK_LCD_PR FOREIGN KEY (ProductionRecordID)
        REFERENCES ProductionRecords(ProductionRecordID),
    CONSTRAINT FK_LCD_Bagging FOREIGN KEY (BaggingCode)
        REFERENCES BaggingLocations(BaggingCode),
    CONSTRAINT FK_LCD_InspectionBy FOREIGN KEY (InspectionBy)
        REFERENCES EmployeeDetails(WindowsLogin),
    
    -- Check Constraints
    CONSTRAINT CHK_BagCounts CHECK (TotalBags >= GoodBags + WasteRejectBags),
    CONSTRAINT CHK_PositiveBags CHECK (TotalBags >= 0 AND GoodBags >= 0 AND WasteRejectBags >= 0)
);

GO
CREATE FUNCTION dbo.GetKpiTargetForUnit (@UnitId INT, @ForDate DATE)
RETURNS @Result TABLE (
    Oee_Target DECIMAL(5,2),
    Waste_Pellet_Target DECIMAL(5,3),
    Waste_Film_Target DECIMAL(5,3),
    GiveAway_Target DECIMAL(6,3),
    GiveAwayMin DECIMAL(6,3),
    GiveAwayMax DECIMAL(6,3),
    TargetYear INT
)
AS
BEGIN
    DECLARE @TargetYear INT = YEAR(@ForDate);
    DECLARE @FoundTarget BIT = 0;
    
    -- 1. ลองหาปีที่ตรงกัน
    INSERT INTO @Result
    SELECT TOP 1 
        Oee_Target,
        Waste_Pellet_Target,
        Waste_Film_Target,
        GiveAway_Target,
        GiveAwayMin,
        GiveAwayMax,
        [Year] AS TargetYear
    FROM KpiTargets
    WHERE UnitId = @UnitId AND [Year] = @TargetYear;
    
    SET @FoundTarget = @@ROWCOUNT;
    
    -- 2. ถ้าไม่เจอปีนั้น ให้ใช้ปีล่าสุดที่มี
    IF @FoundTarget = 0
    BEGIN
        INSERT INTO @Result
        SELECT TOP 1 
            Oee_Target,
            Waste_Pellet_Target,
            Waste_Film_Target,
            GiveAway_Target,
            GiveAwayMin,
            GiveAwayMax,
            [Year] AS TargetYear
        FROM KpiTargets
        WHERE UnitId = @UnitId AND [Year] < @TargetYear
        ORDER BY [Year] DESC;
        
        SET @FoundTarget = @@ROWCOUNT;
    END
    
    -- 3. ถ้ายังไม่มีเลย (unit ใหม่) ใช้ default values
    IF @FoundTarget = 0
    BEGIN
        INSERT INTO @Result VALUES (
            85.00,  -- Default OEE
            0.010,  -- Default Waste Pellet
            0.100,  -- Default Waste Film
            25.000, -- Default Giveaway
            24.900, -- Default Min
            25.100, -- Default Max
            @TargetYear
        );
    END
    
    RETURN;
END
GO

เพิ่มตาราง Subscription สำหรับรายงาน (EmailRecipients, ReportTypes, ReportSubscriptions)

```sql
-- 1. ตาราง EmailRecipients (เก็บผู้รับอีเมล)
CREATE TABLE dbo.EmailRecipients (
    RecipientID INT IDENTITY(1,1) PRIMARY KEY,
    EmployeeName NVARCHAR(100) NULL,
    EmailAddress NVARCHAR(255) NOT NULL,
    Department NVARCHAR(50) NULL,
    Section NVARCHAR(50) NULL,
    IsActive BIT DEFAULT 1,
    CreatedAt DATETIME DEFAULT GETDATE(),
    UpdatedAt DATETIME DEFAULT GETDATE()
);

-- 2. ตาราง ReportTypes (ประเภทของรายงาน)
CREATE TABLE dbo.ReportTypes (
    ReportID INT PRIMARY KEY,
    ReportName NVARCHAR(50),
    Description NVARCHAR(200) NULL
);

INSERT INTO dbo.ReportTypes (ReportID, ReportName, Description) VALUES

(3, 'OEE_QUARTERLY', 'รายงานสรุปรายไตรมาส');

-- 3. ตาราง ReportSubscriptions (การสมัครรับ)
CREATE TABLE dbo.ReportSubscriptions (
    RecipientID INT NOT NULL,
    ReportID INT NOT NULL,
    PRIMARY KEY (RecipientID, ReportID),
    CONSTRAINT FK_Sub_User FOREIGN KEY (RecipientID) REFERENCES dbo.EmailRecipients(RecipientID),
    CONSTRAINT FK_Sub_Report FOREIGN KEY (ReportID) REFERENCES dbo.ReportTypes(ReportID)
);

-- ตัวอย่าง Query ดึงผู้รับรายงานรายเดือน
DECLARE @EmailList NVARCHAR(MAX);
SELECT @EmailList = COALESCE(@EmailList + ';', '') + R.EmailAddress
FROM dbo.EmailRecipients R
JOIN dbo.ReportSubscriptions S ON R.RecipientID = S.RecipientID
WHERE R.IsActive = 1 AND S.ReportID = 2;

SELECT @EmailList AS Recipients;
```
